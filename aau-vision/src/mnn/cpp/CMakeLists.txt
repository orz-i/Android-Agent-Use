cmake_minimum_required(VERSION 3.22.1)
project(aau_mnn_vision LANGUAGES C CXX ASM)

set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_QUANTOOLS OFF CACHE BOOL "" FORCE)
set(MNN_OPENCL OFF CACHE BOOL "" FORCE)
set(MNN_VULKAN OFF CACHE BOOL "" FORCE)
set(MNN_NNAPI OFF CACHE BOOL "" FORCE)
set(MNN_USE_LOGCAT OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${AAU_MNN_SOURCE}" mnn EXCLUDE_FROM_ALL)

add_library(aau_mnn_vision SHARED icon_mnn.cpp)
target_compile_features(aau_mnn_vision PRIVATE cxx_std_17)
target_include_directories(aau_mnn_vision PRIVATE "${AAU_MNN_SOURCE}/include")
target_link_libraries(aau_mnn_vision PRIVATE MNN log)
target_link_options(aau_mnn_vision PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
